    <!-- Course Description -->
    <section class="section fade-in-delay">
        <h2 class="section-title">Course Description</h2>
        <div class="section-content">
            <p>This introductory undergraduate course covers the fundamentals of microeconomics. Topics include supply and demand, market equilibrium, consumer theory, production and the behavior of firms, monopoly, oligopoly, welfare economics, public goods, and externalities.</p>
        </div>
    </section>

    <!-- Topics Covered -->
    <section class="section fade-in-delay">
        <h2 class="section-title">Topics Covered</h2>
        <div class="section-content">
            <h4>Part 1: Foundations</h4>
            <ul>
                <li>Supply and Demand</li>
                <li>Market Equilibrium</li>
                <li>Consumer Theory</li>
            </ul>

            <h4>Part 2: Production & Firms</h4>
            <ul>
                <li>Production Theory</li>
                <li>Behavior of Firms</li>
                <li>Cost Analysis</li>
            </ul>

            <h4>Part 3: Market Structures</h4>
            <ul>
                <li>Perfect Competition</li>
                <li>Monopoly</li>
                <li>Oligopoly</li>
            </ul>

            <h4>Part 4: Market Failures & Policy</h4>
            <ul>
                <li>Welfare Economics</li>
                <li>Public Goods</li>
                <li>Externalities</li>
            </ul>
        </div>
    </section>
